# Character export without an addon - part 1 of 2 (name, level, core attributes, talents).
# In game create a NEW macro (/macro), paste the line below into its body and press
# the macro button (do NOT paste it into chat!). A WEXP1|... line appears in chat -
# copy it into the Hero page.

/run u="player"n=UnitName(u)_,c=UnitClass(u)l=UnitLevel(u)s=UnitStat(u,1)a=UnitStat(u,2)v=UnitStat(u,3)t={}for i=1,3 do _,_,q=GetTalentTabInfo(i)t[i]=q end print("WEXP1|"..n.."|"..c.."|"..l.."|"..s.."|"..a.."|"..v.."|"..table.concat(t,"/"))
