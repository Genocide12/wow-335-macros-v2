# Character export without an addon - part 1 of 2 (character + talents).
# In game create a NEW macro (/macro), paste the line below into its body and press
# the macro button (do NOT paste it into chat!). A WEXP1|... line appears in chat -
# copy it into the Hero page.

/run u="player"n=UnitName(u)_,c=UnitClass(u)l=UnitLevel(u)b,z,v=UnitAttackPower(u)h=UnitHealthMax(u)t={}for i=1,3 do _,_,q=GetTalentTabInfo(i) t[i]=q end print("WEXP1|"..n.."|"..c.."|"..l.."|"..b+z+v.."|"..h.."|"..table.concat(t,"/"))
