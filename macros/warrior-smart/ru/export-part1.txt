# Экспорт персонажа без аддона — часть 1 из 2 (персонаж + таланты).
# Создай в игре НОВЫЙ макрос (/macro), вставь строку ниже в его тело и нажми кнопку макроса
# (НЕ отправляй строку в чат!). В чат выйдет строка WEXP1|... — скопируй её на страницу «Герой».

/run u="player"n=UnitName(u)_,c=UnitClass(u)l=UnitLevel(u)b,z,v=UnitAttackPower(u)h=UnitHealthMax(u)t={}for i=1,3 do _,_,q=GetTalentTabInfo(i) t[i]=q end print("WEXP1|"..n.."|"..c.."|"..l.."|"..b+z+v.."|"..h.."|"..table.concat(t,"/"))
